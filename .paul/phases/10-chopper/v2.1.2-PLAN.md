    ```cpp
    #include <juce_audio_basics/juce_audio_basics.h>

    class Chopper {
    public:
        enum class Shape { sine, square, saw };

        void prepare(double sampleRate);
        void setEnabled(bool e);
        void setDepth(float d);
        void setShape(Shape s);
        // Free rate in Hz (used when sync=false)
        void setRate(float hz);
        // Sync to BPM: division in beats (0.25=1/16, 0.5=1/8, 1=1/4, 2=1/2, 4=1bar)
        void setBPMSync(bool sync, float bpm, float divisionBeats);

        void process(juce::AudioBuffer<float>& buffer);

    private:
        double sampleRate  = 44100.0;
        bool   enabled     = false;
        float  depth       = 1.0f;
        Shape  shape       = Shape::square;
        bool   bpmSync     = true;
        float  rate        = 4.0f;   // Hz
        double phase       = 0.0;    // 0-1
        double phaseInc    = 0.0;    // per sample

        void updatePhaseInc();
        float lfoValue() const;       // returns 0-1
    };
    ```

    **src/Chopper.cpp:**

    `lfoValue()` — returns gain envelope (0=silent, 1=full):
    ```cpp
    float Chopper::lfoValue() const {
        switch (shape) {
            case Shape::sine:
                return 0.5f + 0.5f * (float)std::sin(phase * juce::MathConstants<double>::twoPi);
            case Shape::square:
                return phase < 0.5 ? 1.0f : 0.0f;
            case Shape::saw:
                return 1.0f - (float)phase;  // downward ramp = stuttered fade
        }
        return 1.0f;
    }
    ```

    `process(buffer)`:
    ```cpp
    void Chopper::process(juce::AudioBuffer<float>& buffer) {
        if (!enabled) return;
        const int n = buffer.getNumSamples();
        for (int i = 0; i < n; ++i) {
            const float env = lfoValue();
            const float gain = 1.0f - depth * (1.0f - env);
            for (int ch = 0; ch < buffer.getNumChannels(); ++ch)
                buffer.getWritePointer(ch)[i] *= gain;
            phase += phaseInc;
            if (phase >= 1.0) phase -= 1.0;
        }
    }
    ```
