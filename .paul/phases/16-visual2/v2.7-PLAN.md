    **Step 2 — Main Lissajous with Z + gradient segments**:
    ```cpp
    // Z oscillator: adds 3D depth proportional to l1_timbre
    const float depth = 0.25f + p.l1_timbre * 0.35f;

    // Gradient: divide curve into 16 segments, interpolate color
    const int segments = 16;
    const int ptsPerSeg = N / segments;
    const auto calmColor  = juce::Colour::fromHSV(0.62f, 0.75f, 0.85f, 1.f);
    const auto stormColor = accent;

    // Pre-compute all points
    std::vector<juce::Point<float>> pts(N + 1);
    for (int i = 0; i <= N; ++i) {
        const double t = juce::MathConstants<double>::twoPi * 2.0 * i / N;
        const float z = (float)std::sin(t*(ratio+0.7) + lissPhase*1.4) * 0.4f;
        const float sc = 1.0f / (1.0f + z * depth);
        pts[i] = {cx + (float)std::sin(t)*rx*sc,
                  cy + (float)std::sin(ratio*t+lissPhase)*ry*sc};
    }

    for (int seg = 0; seg < segments; ++seg) {
        const float frac = (float)seg / segments;
        const auto  col  = calmColor.interpolatedWith(stormColor, frac);
        juce::Path  segPath;
        const int   start = seg * ptsPerSeg;
        const int   end   = (seg == segments - 1) ? N : start + ptsPerSeg;
        segPath.startNewSubPath(pts[start].x, pts[start].y);
        for (int i = start + 1; i <= end; ++i) segPath.lineTo(pts[i].x, pts[i].y);
        g.setColour(col.withAlpha(blend * 0.85f));
        g.strokePath(segPath, juce::PathStrokeType(thickness));
    }
    ```

    std::vector<int> sorted(particles.size());
    std::iota(sorted.begin(), sorted.end(), 0);
    std::sort(sorted.begin(), sorted.end(),
              [&](int a, int b){ return particles[a].z < particles[b].z; });

    for (int idx : sorted) {
        const auto& par = particles[idx];
        if (par.life <= 0.f) continue;
        // Depth-based scaling: near=larger/brighter, far=smaller/dimmer
        const float depthScale = 0.6f + (par.z + 0.5f) * 0.8f;  // 0.6 at far, 1.4 at near
        const float alpha = par.life * blend * 0.8f * depthScale;

        const float sy = (par.y + par.z * 0.1f) * h;

        g.fillEllipse(sx - sz*0.5f, sy - sz*0.5f, sz, sz);
