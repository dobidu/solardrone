        : AudioProcessor(BusesProperties()

    {
        engine.setUserParams(userParams);
        // DataFetcher starts its background thread in constructor automatically

    processBlock — poll DataFetcher and update engine when state changes:
    ```cpp
    void SolarDroneAudioProcessor::processBlock(
        juce::AudioBuffer<float>& buffer, juce::MidiBuffer&)
    {
        // Check for new space weather data (cheap: mutex lock + struct copy)
        auto state = fetcher.getState();
        if (state.timestamp != lastTimestamp) {
            lastTimestamp = state.timestamp;
            engine.setSynthParams(SynthParamMapper::map(state, userParams));
        }
        engine.processBlock(buffer);
    }
    ```

    Note: SynthParamMapper::map() is pure — safe on audio thread (no allocation, no locking).
    Note: DataFetcher.getState() acquires a mutex briefly — safe on audio thread for a struct copy.
    Note: lastTimestamp is only accessed on the audio thread — no locking needed.

    Add public getter for status display:
    ```cpp
    // public:
    SpaceWeatherState getLatestSpaceWeatherState() const;
    ```

    Implementation:
    ```cpp
    SpaceWeatherState SolarDroneAudioProcessor::getLatestSpaceWeatherState() const {
        return fetcher.getState();
    }
    ```

  <action>
    Add a small status text overlay to `paint()` in PluginEditor, positioned top-right.
    The text shows: source (live/cached/default), data age in seconds, current Kp.

    Update `paint()`:
    ```cpp
    void SolarDroneAudioProcessorEditor::paint(juce::Graphics& g) {
        g.fillAll(juce::Colours::black);

        // Status overlay — top right, small, semi-transparent
        auto state = processorRef.getLatestSpaceWeatherState();
        juce::String sourceStr;
        juce::Colour statusCol;
        switch (state.source) {
            case SpaceWeatherState::Source::live:
                sourceStr = "live"; statusCol = juce::Colours::limegreen; break;
            case SpaceWeatherState::Source::cached:
                sourceStr = "cached"; statusCol = juce::Colours::yellow; break;
            default:
                sourceStr = "default"; statusCol = juce::Colours::grey; break;
        }
        juce::String statusText = sourceStr
            + "  Kp " + juce::String(state.kp, 1)
            + "  age " + juce::String(state.data_age_s) + "s";

        g.setFont(10.0f);
        g.setColour(statusCol.withAlpha(0.7f));
        g.drawText(statusText,
                   getLocalBounds().reduced(8).removeFromTop(20),
                   juce::Justification::topRight, false);

        // Title
        g.setColour(juce::Colours::white.withAlpha(0.35f));
        g.setFont(11.0f);
        g.drawText("SolarDrone", getLocalBounds().reduced(8),
                   juce::Justification::topLeft, false);
    }
    ```

    Add `#include "SpaceWeatherState.h"` to PluginEditor.cpp (needed for Source enum comparison).
