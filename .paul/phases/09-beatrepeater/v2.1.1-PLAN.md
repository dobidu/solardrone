    **src/TempoTracker.h:**
    ```cpp
    #pragma once
    #include <juce_audio_basics/juce_audio_basics.h>

    class TempoTracker {
    public:
        void prepare(double sampleRate);
        // Call once per processBlock with the full MIDI buffer and block size
        void process(const juce::MidiBuffer& midi, int numSamples);
        void setInternalBPM(float bpm);

        float getCurrentBPM() const;
        // Samples elapsed since last beat boundary (for phase-accurate loop alignment)
        int   getSamplesUntilNextBeat() const;
        bool  isUsingMidiClock() const;

    private:
        static constexpr int kTicksPerBeat = 24;
        static constexpr int kAveragingWindow = 24; // 1 beat worth of ticks

        double sampleRate = 44100.0;
        float  internalBPM = 120.0f;

        // MIDI clock state
        int    tickCount     = 0;
        double lastTickTime  = 0.0;   // in samples
        double currentSample = 0.0;
        double tickIntervals[24] = {}; // ring buffer of intervals
        int    intervalIdx   = 0;
        int    validIntervals = 0;
        double lastMidiClockSample = -1.0;  // -1 = never received

        float  estimatedBPM = 120.0f;
        int    samplesUntilBeat = 0;
        int    beatPeriodSamples = 22050; // 120 BPM at 44100
    };
    ```

    **src/TempoTracker.cpp:**

    `prepare(sr)`: store sampleRate, compute beatPeriodSamples from internalBPM.

    `setInternalBPM(bpm)`: store internalBPM; if not using MIDI, update beatPeriodSamples.

    `process(midi, numSamples)`:
    - Iterate MidiBuffer for `MidiMessage::isMidiClock()` (type == 0xF8)
    - For each tick: compute interval from lastTickTime using sample offset in buffer
    - Store interval in tickIntervals ring buffer
    - After kTicksPerBeat valid intervals: compute BPM = 60 * sampleRate / (sum(intervals)/24)
    - Clamp BPM to [30, 300]
    - Track lastMidiClockSample for fallback detection
    - Advance currentSample by numSamples
    - Decrement samplesUntilBeat by numSamples; when ≤ 0, reset to beatPeriodSamples

    `getCurrentBPM()`:
    ```cpp
    float TempoTracker::getCurrentBPM() const {
        if (isUsingMidiClock()) return estimatedBPM;
        return internalBPM;
    }
    ```

    `isUsingMidiClock()`:
    ```cpp
    bool TempoTracker::isUsingMidiClock() const {
        if (lastMidiClockSample < 0.0) return false;
        return (currentSample - lastMidiClockSample) < sampleRate * 0.5; // 500ms timeout
    }
    ```

    `getSamplesUntilNextBeat()`: return max(0, samplesUntilBeat).

    ```cpp
    #pragma once
    #include <juce_audio_basics/juce_audio_basics.h>
    #include <vector>

    class BeatRepeater {
    public:
        enum class LoopLength { Bar16th, Bar8th, Bar4th, Bar2nd, Bar1, Bar2 };

        void prepare(double sampleRate, int maxBlockSize);
        void setEnabled(bool e);
        void setLoopLength(LoopLength l);
        void setFeedback(float f);
        void setWet(float w);
        void setBeatPeriodSamples(int samples);  // updated each block from TempoTracker

        // Processes in-place. Reads dry input, writes wet output.
        void process(juce::AudioBuffer<float>& buffer);

    private:
        int loopLengthSamples() const;

        double sampleRate    = 44100.0;
        bool   enabled       = false;
        float  feedback      = 0.5f;
        float  wet           = 0.7f;
        int    beatPeriod    = 22050;
        LoopLength loopLen   = LoopLength::Bar4th;

        // Stereo ring buffer — allocated once in prepare()
        std::vector<float> ringL, ringR;
        int writePos         = 0;
        int bufferSize       = 0;  // max 2 bars @ 30 BPM
        bool loopStarted     = false;
    };
    ```

    **src/BeatRepeater.cpp:**

    `prepare(sr, maxBlockSize)`:
    - `sampleRate = sr`
    - Max buffer = 2 bars at 30 BPM: `maxSamples = (int)(sr * 60.0 / 30.0 * 8)` (8 beats = 2 bars)
    - `ringL.assign(maxSamples, 0.f); ringR.assign(maxSamples, 0.f)`
    - `bufferSize = maxSamples; writePos = 0`

    `setBeatPeriodSamples(s)`: `beatPeriod = s`. Recomputes loop length in samples.

    `loopLengthSamples()`:
    ```cpp
    int BeatRepeater::loopLengthSamples() const {
        switch (loopLen) {
            case LoopLength::Bar16th: return beatPeriod / 4;
            case LoopLength::Bar8th:  return beatPeriod / 2;
            case LoopLength::Bar4th:  return beatPeriod;
            case LoopLength::Bar2nd:  return beatPeriod * 2;
            case LoopLength::Bar1:    return beatPeriod * 4;
            case LoopLength::Bar2:    return beatPeriod * 8;
        }
        return beatPeriod;
    }
    ```

    `process(buffer)`:
    ```cpp
    void BeatRepeater::process(juce::AudioBuffer<float>& buffer) {
        auto* dryL = buffer.getReadPointer(0);
        auto* dryR = buffer.getNumChannels() > 1
                     ? buffer.getReadPointer(1) : dryL;
        auto* outL = buffer.getWritePointer(0);
        auto* outR = buffer.getNumChannels() > 1
                     ? buffer.getWritePointer(1) : outL;
        const int n = buffer.getNumSamples();
        const int loopLen = loopLengthSamples();

        for (int i = 0; i < n; ++i) {
            // Write dry + fedback loop into ring buffer
            const int readPos = (writePos - loopLen + bufferSize) % bufferSize;
            const float loopedL = ringL[readPos];
            const float loopedR = ringR[readPos];

            ringL[writePos] = dryL[i] + loopedL * feedback;
            ringR[writePos] = dryR[i] + loopedR * feedback;

            // Mix output
            if (enabled) {
                outL[i] = dryL[i] * (1.0f - wet) + loopedL * wet;
                outR[i] = dryR[i] * (1.0f - wet) + loopedR * wet;
            }
            // (if disabled, buffer unchanged = transparent)

            writePos = (writePos + 1) % bufferSize;
        }
    }
    ```

    Note: ring buffer writes continuously even when disabled — so enabling doesn't start from silence.
    Note: no mutex needed — all calls from audio thread only.
