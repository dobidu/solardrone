    **CMakeLists.txt** (root):
    ```cmake
    cmake_minimum_required(VERSION 3.22)
    project(SolarDrone VERSION 0.1.0)

    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)

    include(FetchContent)
    FetchContent_Declare(
      JUCE
      GIT_REPOSITORY https://github.com/juce-framework/JUCE.git
      GIT_TAG 8.0.4
      GIT_SHALLOW TRUE
    )
    FetchContent_MakeAvailable(JUCE)

    juce_add_plugin(SolarDrone
        PLUGIN_MANUFACTURER_CODE Sldr
        PLUGIN_CODE Sldr
        BUNDLE_ID "com.solardrone.SolarDrone"
        PLUGIN_AU_EXPORT_PREFIX SolarDroneAU
        FORMATS VST3 AU Standalone
        PRODUCT_NAME "SolarDrone"
        PLUGIN_MANUFACTURER_NAME "SolarDrone"
        IS_SYNTH FALSE
        NEEDS_MIDI_INPUT FALSE
        NEEDS_MIDI_OUTPUT FALSE
        IS_MIDI_EFFECT FALSE
        EDITOR_WANTS_KEYBOARD_FOCUS FALSE
        COPY_PLUGIN_AFTER_BUILD FALSE
    )

    target_sources(SolarDrone PRIVATE
        src/PluginProcessor.cpp
        src/PluginEditor.cpp
    )

    target_compile_definitions(SolarDrone PUBLIC
        JUCE_WEB_BROWSER=0
        JUCE_USE_CURL=0
        JUCE_VST3_CAN_REPLACE_VST2=0
        JUCE_DISPLAY_SPLASH_SCREEN=0
    )

    target_link_libraries(SolarDrone
        PRIVATE
            juce::juce_audio_utils
            juce::juce_dsp
        PUBLIC
            juce::juce_recommended_config_flags
            juce::juce_recommended_lto_flags
            juce::juce_recommended_warning_flags
    )
    ```

    Note: AU target is only built on macOS; JUCE CMake skips it on Linux/Windows automatically.

  <name>Task 2: GitHub Actions CI matrix — build on all 3 OSes</name>
  <files>.github/workflows/build.yml</files>
  <action>
    Create `.github/workflows/build.yml`:
    ```yaml
    name: Build

    on:
      push:
        branches: [main]
      pull_request:
        branches: [main]

    jobs:
      build:
        name: Build (${{ matrix.os }})
        runs-on: ${{ matrix.os }}
        strategy:
          fail-fast: false
          matrix:
            os: [ubuntu-latest, macos-latest, windows-latest]

        steps:
          - uses: actions/checkout@v4

          - name: Install Linux system dependencies
            if: runner.os == 'Linux'
            run: |
              sudo apt-get update
              sudo apt-get install -y \
                libasound2-dev \
                libfreetype6-dev \
                libx11-dev \
                libxcomposite-dev \
                libxext-dev \
                libxinerama-dev \
                libxrandr-dev \
                libxrender-dev \
                libglu1-mesa-dev \
                mesa-common-dev \
                libwebkit2gtk-4.0-dev \
                libcurl4-openssl-dev

          - name: Configure CMake
            run: cmake -B build -DCMAKE_BUILD_TYPE=Release

          - name: Build (Standalone + VST3)
            run: cmake --build build --config Release --target SolarDrone_Standalone SolarDrone_VST3
    ```

    Notes:
    - `fail-fast: false` lets all 3 OSes report independently.
    - AU target not explicitly built in CI (requires code signing in Phase 7).
    - Linux dependencies are the standard JUCE set for Ubuntu.
    - FetchContent will clone JUCE on first CI run (~1-2 min); subsequent runs use GitHub Actions cache (add caching if CI time becomes an issue in Phase 7).
