    btnTabVisual.setButtonText("VISUAL");
    btnTabVisual.setColour(juce::TextButton::buttonColourId, juce::Colour(0xff1a2a4a));
    btnTabVisual.setColour(juce::TextButton::textColourOffId, juce::Colours::white);

        repaint();
    };
    addAndMakeVisible(btnTabVisual);

    btnTabResonators.setButtonText("RESONATORS");
    btnTabResonators.setColour(juce::TextButton::buttonColourId, juce::Colour(0xff1a2a4a));
    btnTabResonators.setColour(juce::TextButton::textColourOffId, juce::Colours::white);
    btnTabResonators.onClick = [this]() {
        showResonators = true;
        visualRenderer.setVisible(false);
        resonatorPanel.setVisible(true);
        sunDisc.setVisible(false);
        repaint();
    };
    addAndMakeVisible(btnTabResonators);
    addAndMakeVisible(resonatorPanel);
    resonatorPanel.setVisible(false);  // hidden by default
