    **B. Bz spark-line** (drawn at bottom edge of visual, 8px tall strip):
    ```cpp
    const float sparkH = 8.0f;
    const float sparkY = getHeight() - sparkH - 1.0f;
    const float sparkW = (float)getWidth();
    const int   nSamples = 120;

    g.setColour(juce::Colours::black.withAlpha(0.5f));
    g.fillRect(0.f, sparkY, sparkW, sparkH);

    // Draw Bz waveform
    juce::Path sparkPath;
    for (int i = 0; i < nSamples; ++i) {
        const int idx = (bzHistoryIdx - nSamples + i + 120) % 120;
        const float bz  = bzHistory[idx];
        const float x   = sparkW * i / (float)nSamples;
        const float norm = juce::jlimit(-1.f, 1.f, bz / 30.f); // ±30nT range
        const float y   = sparkY + sparkH * 0.5f - norm * sparkH * 0.45f;
        if (i == 0) sparkPath.startNewSubPath(x, y);
        else        sparkPath.lineTo(x, y);
    }
    // Color by most-recent Bz
    const float latestBz = bzHistory[(bzHistoryIdx - 1 + 120) % 120];
    g.setColour(latestBz < 0
        ? juce::Colours::orangered.withAlpha(0.8f)
        : ColourScheme::kpToAccent(cachedWeatherState.kp).withAlpha(0.6f));
    g.strokePath(sparkPath, juce::PathStrokeType(1.2f));
    ```
