    ```cpp
    #pragma once
    #include <juce_gui_basics/juce_gui_basics.h>

    struct ColourScheme {
        // Returns accent colour for current Kp (0-9).
        // kp 0-2: deep blue  #1a3a6e → teal → amber → red at kp 7-9
        static juce::Colour kpToAccent(float kp) {
            kp = std::max(0.f, std::min(9.f, kp));
            if (kp < 2.f)
                return juce::Colour::fromHSV(0.62f, 0.75f, 0.85f, 1.f);
            if (kp < 5.f) {
                const float t = (kp - 2.f) / 3.f;
                return juce::Colour::fromHSV(0.62f - t * 0.17f, 0.75f, 0.9f, 1.f);
            }
            if (kp < 7.f) {
                const float t = (kp - 5.f) / 2.f;
                return juce::Colour::fromHSV(0.45f - t * 0.34f, 0.85f, 1.0f, 1.f);
            }
            const float t = (kp - 7.f) / 2.f;
            return juce::Colour::fromHSV(0.11f - t * 0.11f, 0.9f, 1.0f, 1.f);
        }

        // Background panel colour — very dark, tinted by Kp
        static juce::Colour background(float kp) {
            return kpToAccent(kp).withBrightness(0.08f).withSaturation(0.4f);
        }

        // Grid line colour for space-grid background
        static juce::Colour grid(float kp) {
            return kpToAccent(kp).withAlpha(0.06f);
        }
    };
    ```

    `paint(g)`:
    ```cpp
    void SunDisc::paint(juce::Graphics& g) {
        const float vol    = getVolume();
        const bool  on     = getDroneOn();
        const auto  accent = ColourScheme::kpToAccent(currentKp);
        const float w = (float)getWidth(), h = (float)getHeight();
        const float r = std::min(w, h) * 0.5f;
        const juce::Point<float> centre{w * 0.5f, h * 0.5f};

        // Outer rim (Kp glow)
        g.setColour(accent.withAlpha(on ? 0.9f : 0.3f));
        g.drawEllipse(centre.x - r + 2, centre.y - r + 2, (r - 2) * 2, (r - 2) * 2, 3.0f);

        // Volume arc fill (clockwise from bottom)
        if (on && vol > 0.01f) {
            const float startAngle = juce::MathConstants<float>::pi;   // bottom
            const float endAngle   = startAngle + vol * juce::MathConstants<float>::twoPi;
            juce::Path arc;
            arc.addArc(centre.x - r * 0.75f, centre.y - r * 0.75f,
                       r * 1.5f, r * 1.5f, startAngle, endAngle, true);
            g.setColour(accent.withAlpha(0.35f));
            g.strokePath(arc, juce::PathStrokeType(4.0f));
        }

        // Inner disc
        const float ir = r * 0.65f;
        g.setColour(juce::Colour(0xff050a14).withAlpha(on ? 0.95f : 0.6f));
        g.fillEllipse(centre.x - ir, centre.y - ir, ir * 2, ir * 2);

        // Symbol
        g.setFont(r * 0.4f);
        g.setColour(accent.withAlpha(on ? 0.95f : 0.3f));
        g.drawText(on ? "☀" : "○", getLocalBounds(), juce::Justification::centred, false);
    }
    ```

    void SolarDroneAudioProcessorEditor::paint(juce::Graphics& g) {
        // Update Kp for colour scheme
        currentKp = processorRef.getLatestSpaceWeatherState().kp;
        const auto bg     = ColourScheme::background(currentKp);
        const auto grid   = ColourScheme::grid(currentKp);
        const auto accent = ColourScheme::kpToAccent(currentKp);

        // Background
        g.fillAll(bg);

        // Space grid (subtle lines every 40px)
        g.setColour(grid);
        for (int x = 0; x < 900; x += 40)
            g.drawVerticalLine(x, 0, 600);
        for (int y = 0; y < 600; y += 40)
            g.drawHorizontalLine(y, 0, 900);

        // Right panel background — slightly lighter
        g.setColour(accent.withAlpha(0.04f));
        g.fillRect(580, 0, 320, 520);

        // Bottom strip background
        g.setColour(juce::Colour(0xff080e1a));
        g.fillRect(0, 520, 900, 80);

        // Panel dividers
        g.setColour(accent.withAlpha(0.15f));
        g.drawVerticalLine(579, 0, 520);

        // "REPEATER" / "CHOPPER" labels in strip
