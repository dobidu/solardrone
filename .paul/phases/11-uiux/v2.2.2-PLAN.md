    **src/MacroOrb.cpp:**

    Corner presets:
    ```cpp
    const MacroOrb::Preset MacroOrb::kCorners[4] = {
        {0.0f, 0.5f, 0.0f, 0.0f},   // TL: Solar Wind (L1 pure drone)
        {1.0f, 1.0f, 0.0f, 0.0f},   // TR: Geomagnetic (L2 pure drone)
        {0.2f, 0.4f, 0.85f, 0.3f},  // BL: Ghost Loop (L1 + heavy repeat)
        {0.9f, 1.0f, 0.8f, 1.0f},   // BR: Storm Gate (L2 + repeat + chop)
    };
    ```

    `setKp(kp)`: store, repaint.

    `applyBlend()`:
    ```cpp
    const float x = cursor.x, y = cursor.y;
    // y=0 at top, y=1 at bottom
    const float wTL = (1-x)*(1-y), wTR = x*(1-y);
    const float wBL = (1-x)*y,     wBR = x*y;

    auto blend4 = [&](auto fn) {
        return fn(kCorners[0])*wTL + fn(kCorners[1])*wTR
             + fn(kCorners[2])*wBL + fn(kCorners[3])*wBR;
    };
    auto setP = [&](const char* id, float v) {
        if (auto* p = apvts.getParameter(id))
            p->setValueNotifyingHost(p->convertTo0to1(v));
    };
    setP("layer_balance",  blend4([](const Preset& c){ return c.balance;   }));
    setP("dynamics_range", blend4([](const Preset& c){ return c.dynamics;  }));
    setP("repeater_wet",   blend4([](const Preset& c){ return c.repWet;    }));
    setP("chopper_depth",  blend4([](const Preset& c){ return c.chopDepth; }));
    ```

    `mouseDown(e)`: convert to normalized position; call applyBlend(); repaint().
    `mouseDrag(e)`: same as mouseDown.

    Helper in both:
    ```cpp
    cursor.x = std::max(0.f, std::min(1.f, e.position.x / (float)getWidth()));
    cursor.y = std::max(0.f, std::min(1.f, e.position.y / (float)getHeight()));
    applyBlend();
    repaint();
    ```

    **VisualRenderer.h**: add ghost overlay state:
    ```cpp

    bool           frozen        = false;
    bool           wasFreeze     = false;  // edge detect
    juce::Image    frozenSnapshot;
    ```

    **VisualRenderer.cpp — timerCallback()**: detect freeze activation edge:
    ```cpp
    // After polling processor:
    bool nowFrozen = false;
    if (processor != nullptr)
        nowFrozen = *processor->apvts.getRawParameterValue("freeze_on") > 0.5f;

    if (nowFrozen && !wasFreeze) {
        // Just activated — take snapshot of current visual
        frozenSnapshot = createComponentSnapshot(getLocalBounds(), 1.0f);
    }
    wasFreeze = nowFrozen;
    frozen    = nowFrozen;
    ```

    **VisualRenderer::paint()**: draw ghost overlay at end:
    ```cpp
    // After all layers drawn, before status text:
    if (frozen && frozenSnapshot.isValid()) {
        g.setOpacity(0.25f);
        g.drawImage(frozenSnapshot, getLocalBounds().toFloat());
        g.setOpacity(1.0f);
        // "FROZEN" badge
        g.setColour(juce::Colours::cyan.withAlpha(0.7f));
        g.setFont(10.0f);
        g.drawText("◼ FROZEN", getLocalBounds().reduced(8).removeFromTop(18),
                   juce::Justification::topLeft, false);
    }
    ```

    Note: `createComponentSnapshot()` runs on message thread (from timerCallback) — safe.
    Note: `processor->apvts.getRawParameterValue("freeze_on")` is an atomic read — safe from message thread.
